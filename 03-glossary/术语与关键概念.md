```dataviewjs
const pages = dv.pages('"02-terms"'); // 改成你的术语笔记文件夹

const container = dv.container;

// 头部
const header = container.createEl('header', { cls: 'reference-heading' });
header.innerHTML = `
    <p>概念索引</p>
    <h2>术语与关键概念</h2>
    <div class="section-deck">按定义理解概念，沿阅读定位返回相关论述。</div>
`;

// 控件容器
const controls = container.createEl('div', { cls: 'gloss-controls' });

// 搜索框
const searchLabel = controls.createEl('label', { cls: 'field' });
searchLabel.createEl('span', { text: '查找术语' });
const searchInput = searchLabel.createEl('input', {
    attr: { type: 'search', placeholder: '输入中文、英文或关键词', autocomplete: 'off', id: 'gloss-search' }
});

// 分类按钮
const filterBar = controls.createEl('div', { cls: 'filter-bar', attr: { role: 'group', 'aria-label': '术语分类' } });
const categories = ['全部', ...new Set(pages.map(p => p.category).filter(Boolean))];
let activeCategory = '全部';

const categoryButtons = {};
categories.forEach(cat => {
    const isActive = cat === activeCategory;
    const btn = filterBar.createEl('button', {
        text: cat,
        cls: 'filter' + (isActive ? ' active' : ''),
        attr: { 'data-category': cat, 'aria-pressed': isActive ? 'true' : 'false' }
    });
    btn.addEventListener('click', () => {
        activeCategory = cat;
        Object.values(categoryButtons).forEach(b => {
            b.classList.remove('active');
            b.setAttribute('aria-pressed', 'false');
        });
        btn.classList.add('active');
        btn.setAttribute('aria-pressed', 'true');
        renderTable();
    });
    categoryButtons[cat] = btn;
});

// 结果计数
const countEl = controls.createEl('p', { cls: 'result-count', attr: { role: 'status' } });

// 表格容器
const tableWrap = container.createEl('div', { cls: 'gloss-table-wrap' });

// 空结果提示
const emptyEl = container.createEl('p', {
    cls: 'gloss-empty',
    text: '没有找到匹配概念。可以缩短关键词，或选择“全部”。',
    attr: { hidden: '' }
});

function renderTable() {
    const searchTerm = searchInput.value.toLowerCase();
    const filtered = pages.filter(p => {
        const term = (p.term || '').toString().toLowerCase();
        const english = (p.english || '').toString().toLowerCase();
        const def = (p.definition || '').toString().toLowerCase();
        const context = (p.context || '').toString().toLowerCase();
        const sectionLabel = (p.section_label || '').toString().toLowerCase();
        const searchTarget = `${term} ${english} ${def} ${context} ${sectionLabel}`;
        const matchesSearch = searchTarget.includes(searchTerm);
        const matchesCategory = activeCategory === '全部' || p.category === activeCategory;
        return matchesSearch && matchesCategory;
    });

    countEl.textContent = `共 ${filtered.length} 个概念`;
    tableWrap.innerHTML = '';
    if (filtered.length === 0) {
        emptyEl.hidden = false;
        return;
    }
    emptyEl.hidden = true;

    const table = tableWrap.createEl('table', { cls: 'gloss-table' });
    const thead = table.createEl('thead');
    const headerRow = thead.createEl('tr');
    ['术语', '定义与本文语境', '阅读定位'].forEach(th => headerRow.createEl('th', { text: th }));

    const tbody = table.createEl('tbody');
    filtered.forEach(p => {
        const tr = tbody.createEl('tr', { attr: { 'data-category': p.category || '' } });

        // 第一列：术语名称
        const th = tr.createEl('th', { attr: { scope: 'row' } });
        th.createEl('span', { text: p.term });
        if (p.english) th.createEl('small', { text: p.english });
        th.createEl('em', { text: p.category || '' });

        // 第二列：定义与语境（无引用）
        const td = tr.createEl('td');
        td.createEl('p', { text: p.definition });
        if (p.context) {
            td.createEl('p', { cls: 'term-context', text: p.context });
        }
        
        // 第三列：阅读定位（支持多个双链）
		const td2 = tr.createEl('td');
		if (p.section_label) {
		    // 确保是数组
		    const labels = Array.isArray(p.section_label) ? p.section_label : [p.section_label];
		    // 用分隔符连接渲染后的链接
		    const linkContainer = td2.createEl('div', { cls: 'read-locations' });
		    labels.forEach((label, index) => {
		        if (index > 0) {
		            // 在两个链接之间添加分隔符
		            linkContainer.createEl('span', { text: ' · ', cls: 'link-separator' });
		        }
		        dv.span(label, { container: linkContainer });
		    });
		} else {
		    td2.createEl('span', { text: '—' });
		}
    });
}

searchInput.addEventListener('input', renderTable);
renderTable();
```
